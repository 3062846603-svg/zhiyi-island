一、核心技术亮点

1. PostgreSQL pgvector向量搜索
实现细节：
  - 使用 pgvector 扩展存储1024维向量
  - 笔记表 note.embedding 字段类型为 vector(1024)
  - 使用 HNSW 索引加速相似度搜索：CREATE INDEX idx_note_embedding ON note USING hnsw (embedding vector_cosine_ops)
  - 自定义 VectorTypeHandler 处理 Java float[] 与 PostgreSQL vector 类型的转换
  - 使用余弦距离操作符 <=> 计算相似度
技术难点：
  - MyBatis 不支持 PostgreSQL 的 vector 类型，需自定义 TypeHandler
  - 向量字符串格式为 [0.1,0.2,...]，需手动解析和构建
解决方案：
  // VectorTypeHandler.java 
  private float[] parseVector(String value) { 
      value = value.replaceAll("[\\[\\]]", ""); 
      String[] parts = value.split(","); 
      float[] result = new float[parts.length]; 
      for (int i = 0; i < parts.length; i++) { 
          result[i] = Float.parseFloat(parts[i].trim()); 
      } 
      return result; 
  }

  SELECT *, 1 - (embedding <=> #{embedding}::vector) as similarity 
  FROM note 
  WHERE user_id = #{userId} AND embedding IS NOT NULL 
  ORDER BY embedding <=> #{embedding}::vector 
  LIMIT #{limit}

  // 嵌入API：调用 /embeddings，返回 data[0].embedding 数组 
  public float[] generateEmbedding(String text) { 
      String url = baseUrl + "/embeddings"; 
      requestBody.put("model", embeddingModel); 
      requestBody.put("input", text); 
      requestBody.put("encoding_format", "float"); 
      // 统一解析：从 data[0].embedding 提取向量 
      Map<String, Object> response = restTemplate.postForObject(url, entity, Map.class); 
      List<Map<String, Object>> dataList = (List<Map<String, Object>>) response.get("data"); 
      List<Number> embeddingList = (List<Number>) dataList.get(0).get("embedding"); 
      float[] embedding = new float[embeddingList.size()]; 
      for (int i = 0; i < embeddingList.size(); i++) { 
          embedding[i] = embeddingList.get(i).floatValue(); 
      } 
      return embedding; 
  }

3. JWT + Redis Token管理
实际配置：
  jwt: 
    secret: zhiyiIslandSecretKey2024VeryLongSecretKeyForJWTTokenGeneration 
    expiration: 86400000 # 24小时
实现细节：
  - JWT Token 存储用户ID、用户名、过期时间
  - Redis 双重存储：
    - token:{token} → userId（单点验证）
    - user:tokens:{userId} → Set<token>（支持全设备登出）
  - Token 过期时间与 Redis TTL 同步（24小时）
技术难点：
  - JWT 无状态，无法主动失效
  - 用户多设备登录时需要管理所有 Token
解决方案：
  // 保存Token时同时维护两个映射 
  public void saveToken(String token, Long userId) { 
      String tokenKey = TOKEN_PREFIX + token; 
      String userTokensKey = USER_TOKENS_PREFIX + userId; 
      redisTemplate.opsForValue().set(tokenKey, userId, 86400000, TimeUnit.MILLISECONDS); 
      redisTemplate.opsForSet().add(userTokensKey, token); 
      redisTemplate.expire(userTokensKey, 86400000, TimeUnit.MILLISECONDS); 
  } 
  // 全设备登出 
  public void invalidateAllTokens(Long userId) { 
      String userTokensKey = USER_TOKENS_PREFIX + userId; 
      Set<Object> tokens = redisTemplate.opsForSet().members(userTokensKey); 
      for (Object token : tokens) { 
          redisTemplate.delete(TOKEN_PREFIX + token); 
      } 
      redisTemplate.delete(userTokensKey); 
  }

6. 异步向量嵌入生成
实现细节：
  - 笔记创建后异步生成向量嵌入
  - 使用 @Async 注解实现异步调用
  - 向量生成失败不影响笔记保存
  - 支持批量补全缺失向量的笔记
技术难点：
  - 向量生成耗时较长（约2-5秒）
  - 不能阻塞用户操作
  - 需要处理生成失败的情况
解决方案：
  @Async 
  public void generateEmbeddingAsync(Long noteId, String content) { 
      try { 
          float[] embedding = aiProviderService.generateEmbedding(content); 
          noteMapper.updateEmbedding(noteId, embedding); 
          log.info("向量生成成功，笔记ID：{}，维度：{}", noteId, embedding.length); 
      } catch (Exception e) { 
          log.error("向量生成失败，笔记ID：{}", noteId); 
          // 失败不影响笔记，可后续补全 
      } 
  } 
  // 批量补全 
  public void batchGenerateEmbedding(Long userId) { 
      List<Note> notes = noteMapper.selectNotesWithoutEmbedding(userId, 10); 
      for (Note note : notes) { 
          generateEmbeddingAsync(note.getId(), note.getContent()); 
      } 
  }

      } 
      @ExceptionHandler(MethodArgumentNotValidException.class) 
      public Result<Void> handleValidationException(MethodArgumentNotValidException e) { 
          String message = e.getBindingResult().getFieldErrors().stream()

  const viewBox = computed(() => { 
    const width = 1000, height = 700 
    const scaledWidth = width / scale.value 
    const scaledHeight = height / scale.value 
    const x = (width - scaledWidth) / 2 - translateX.value 
    const y = (height - scaledHeight) / 2 - translateY.value 
    return `${x} ${y} ${scaledWidth} ${scaledHeight}` 
  })
  // 节点布局：核心居中 + 角度均匀分布 
  nodes.value = result.nodes.map((node, index) => { 
    const isCore = index === 0 
    const baseRadius = isCore ? 0 : 160 + (index % 3) * 80 
    const angle = (index / (nodeCount - 1)) * Math.PI * 2 - Math.PI / 2 
    return { 
      x: isCore ? centerX : centerX + Math.cos(angle) * baseRadius + jitterX, 
      y: isCore ? centerY : centerY + Math.sin(angle) * baseRadius + jitterY, 
    } 
  })

  // 防止多次重定向 
  let isRedirecting = false 
  api.interceptors.response.use( 
    (response) => response.data, 
    (error) => { 
      if (error.response?.status === 401 || error.response?.status === 403) { 
        sessionStorage.removeItem('auth_state') 
        if (!isRedirecting && window.location.pathname !== '/login') { 
          isRedirecting = true 
          window.location.href = '/login' 
        } 
      } 
      if (error.code === 'ERR_NETWORK') { 
        return Promise.reject({ message: '服务暂不可用，请稍后重试' }) 
      } 
      if (error.message?.includes('timeout')) { 
        return Promise.reject({ message: '请求超时，请稍后重试' }) 
      } 
      return Promise.reject({ message: '网络连接异常，请检查网络' }) 
    } 
  )

  })

  onBeforeUnmount(() => {

    if (editor) editor.destroy()

  const uploadFiles = async (files) => { 
    const remainingSlots = props.maxCount - images.value.length 
    const filesToUpload = files.slice(0, remainingSlots) 
    for (const file of filesToUpload) { 
      if (!file.type.startsWith('image/')) { emit('error', '不是图片文件'); continue } 
      if (file.size > props.maxSize) { emit('error', '大小超过限制'); continue } 
      const result = await fileApi.uploadNoteImage(file) 
      if (result.code === 200) images.value = [...images.value, result.data.url] 
    } 
  }

7. 路由守卫与权限控制
实现细节：
  - 使用 Vue Router 5 的 beforeEach 全局守卫
  - 从 sessionStorage 读取Token判断登录状态
  - 未登录用户访问需认证页面自动跳转登录页
  - 已登录用户访问登录页自动跳转首页
  - 法律页面（用户协议、隐私政策等）无需登录即可访问
解决方案：
  router.beforeEach((to) => { 
    const authState = sessionStorage.getItem('auth_state') 
    const token = authState ? JSON.parse(authState).token : null 
    const requiresAuth = to.meta.requiresAuth !== false 
    const isLoginPage = to.path === '/login' 
    if (requiresAuth && !token && !isLoginPage) return '/login' 
    if (token && isLoginPage) return '/home' 
  })
